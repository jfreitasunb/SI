%colocar um ifdraft, se for incluir de fato um bmp sem keepaspectratio=true
%caso contrário, incluir eps com keepaspectratio=true
%\DeclareGraphicsExtension{bmp eps}
%natwidth, natheight

\def\setbmp#1#2#3#4{\vskip#3\relax\noindent\hskip#1\relax
    \includegraphics[natwidth=#2, natheight=#3]{#4}
}

\def\centerbmp#1#2#3{\centerline{\hbox to#1{
    \includegraphics[natwidth=#1, natheight=#2]{#3}
%\hfil
}}}

\def\seteps#1#2#3#4{\vskip#3\relax\noindent\hskip#1\relax
    \includegraphics[natwidth=#2, natheight=#3]{#4}
}

\def\centereps#1#2#3{
\relax
    \centerline{
       \hbox to#1{\includegraphics[width=#1, keepaspectratio=true]{#3}}
    }
}

\def\centerepsaltura#1#2#3{
\relax
    \centerline{
       \hbox to#1
       {\includegraphics[height=#2, keepaspectratio=true]{#3}}
    }
}
